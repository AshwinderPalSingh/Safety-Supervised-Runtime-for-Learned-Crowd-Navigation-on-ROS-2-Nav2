cmake_minimum_required(VERSION 3.16)
project(crowd_nav_safety_supervisor)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()

find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(crowd_nav_observation REQUIRED)
find_package(crowd_nav_policy_adapters REQUIRED)

rosidl_generate_interfaces(${PROJECT_NAME}
  "msg/InterventionEvent.msg"
  DEPENDENCIES builtin_interfaces
)
# Message generation + the C++ library that constructs/publishes it live in this one package
# (S4.8.7's InterventionEvent has no other producer) - rosidl_get_typesupport_target is the
# documented way to depend on interfaces generated by this SAME CMakeLists.txt, since a target
# can't rosidl_generate_interfaces its own dependency the normal cross-package way.
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} "rosidl_typesupport_cpp")

# SHARED - see crowd_nav_perception/CMakeLists.txt's comment: this workspace's CMake default is
# static, which lacks -fPIC and can't be linked into crowd_nav_controller's pluginlib .so.
add_library(${PROJECT_NAME}_lib SHARED
  src/safety_supervisor.cpp
)
target_include_directories(${PROJECT_NAME}_lib PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
)
ament_target_dependencies(${PROJECT_NAME}_lib
  rclcpp nav2_costmap_2d nav_msgs crowd_nav_observation crowd_nav_policy_adapters
)
target_link_libraries(${PROJECT_NAME}_lib "${cpp_typesupport_target}")

install(
  DIRECTORY include/
  DESTINATION include
)
install(
  TARGETS ${PROJECT_NAME}_lib
  EXPORT ${PROJECT_NAME}_libTargets
  LIBRARY DESTINATION lib
  ARCHIVE DESTINATION lib
  RUNTIME DESTINATION bin
)

ament_export_targets(${PROJECT_NAME}_libTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(
  rclcpp nav2_costmap_2d nav_msgs crowd_nav_observation crowd_nav_policy_adapters
  rosidl_default_runtime
)
ament_export_include_directories(include)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_safety_supervisor test/test_safety_supervisor.cpp)
  target_link_libraries(test_safety_supervisor ${PROJECT_NAME}_lib "${cpp_typesupport_target}")
  ament_target_dependencies(test_safety_supervisor
    rclcpp nav2_costmap_2d nav_msgs crowd_nav_observation crowd_nav_policy_adapters
  )
endif()

ament_package()
